import { useMemo, useState } from 'react';

/**

// LibellÃ©s lisibles des ENUM rÃ©els de la base.
const TYPE_LABELS: Record<string, string> = {
  academique: 'AcadÃ©mique',
  vie_scolaire: 'Vie scolaire',
  finance: 'Finance',
  autre: 'Autre',
};

const CIBLE_LABELS: Record<string, string> = {
  tous: 'Tous les Ã©tudiants',
  filiere: 'FiliÃ¨re',
  classe: 'Classe',
  etudiant: 'Ã‰tudiant',
};

const labelType = (type: string) => TYPE_LABELS[type] || type || 'Autre';

const getTypeColor = (type: string) => {
  switch (type) {
    case 'academique': return 'bg-blue-100 text-blue-800 border-blue-300';
    case 'vie_scolaire': return 'bg-purple-100 text-purple-800 border-purple-300';
    case 'finance': return 'bg-orange-100 text-orange-800 border-orange-300';
    default: return 'bg-gray-100 text-gray-800 border-gray-300';
  }
};

const getStatutColor = (statut: string) => {
  switch (statut) {
    case 'Envoyee': return 'bg-green-100 text-green-800';
    case 'Programmee': return 'bg-yellow-100 text-yellow-800';
    default: return 'bg-gray-100 text-gray-800';
  }
};

// Date_Notif / Scheduled_At / Sent_At sont des DATETIME MySQL.
const formatDate = (value: string) => {
  if (!value) return 'â€”';
  const date = new Date(String(value).replace(' ', 'T'));
  if (Number.isNaN(date.getTime())) return String(value);
  return date.toLocaleString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' });
};

/** Destinataire rÃ©el, reconstruit depuis les jointures du backend. */
const destinatairesTexte = (notif: any) => {
  switch (notif.cible) {
    case 'filiere': return notif.filiere || 'FiliÃ¨re introuvable';
    case 'classe': return notif.classe || 'Classe introuvable';
    case 'etudiant': return notif.etudiant || 'Ã‰tudiant introuvable';
    default: return 'Tous les Ã©tudiants';
  }
};

export default function Notifications() {
  const { data: notifications, loading, error, reload } = useApiData<any[]>(notificationsService.getAll, []);
  const { data: filieres } = useApiData<any[]>(filieresService.getAll, []);
  const { data: classes } = useApiData<any[]>(classesService.getAll, []);
  const { data: etudiants } = useApiData<any[]>(etudiantsService.getAll, []);
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedType, setSelectedType] = useState('Tous');
  const [selectedStatut, setSelectedStatut] = useState('Tous');
  const [modal, setModal] = useState<'create' | null>(null);
  const [submitError, setSubmitError] = useState('');
  const [submitting, setSubmitting] = useState(false);

  // Les donnÃ©es viennent de la BD : filtrage effectuÃ© sur les vrais champs.
  const filteredNotifications = useMemo(() => {

        (value) => String(value || '').toLowerCase().includes(q),

      return matchesSearch && matchesType && matchesStatut;

    try {
      await notificationsService.create({
        titre: values.titre.trim(),
        message: values.message.trim(),
        type: values.type,

        idFiliere: values.cible === 'filiere' ? Number(values.idFiliere) : null,

        idEtudiant: values.cible === 'etudiant' ? Number(values.idEtudiant) : null,

      setModal(null);

    } catch (requestError: any) {
      setSubmitError(requestError?.response?.data?.error || requestError?.message || 'Impossible dâ€™envoyer la notification.');
    } finally { setSubmitting(false); }
  };

  const removeNotification = async (notif: any) => {
    if (!window.confirm(`Supprimer la notification Â« ${notif.titre} Â» ?`)) return;
    try { await notificationsService.delete(notif.id); await reload(); }

  };

  return (
    <div className="space-y-6">

      )}

        <div className="bg-white p-6 rounded-lg shadow-sm border border-gray-200">
          <div className="flex items-center justify-between">
            <div>

            </div>
            <Bell className="h-8 w-8 text-blue-500" />
          </div>
        </div>
        <div className="bg-white p-6 rounded-lg shadow-sm border border-gray-200">

            <div>
              <p className="text-sm font-medium text-gray-600">EnvoyÃ©es</p>
              <p className="text-2xl font-bold text-green-600 mt-1">{totalEnvoyees}</p>

          </div>
        </div>
        <div className="bg-white p-6 rounded-lg shadow-sm border border-gray-200">
          <div className="flex items-center justify-between">
            <div>
              <p className="text-sm font-medium text-gray-600">ProgrammÃ©es</p>
              <p className="text-2xl font-bold text-yellow-600 mt-1">{totalProgrammees}</p>
            </div>
            <Clock className="h-8 w-8 text-yellow-500" />
          </div>
        </div>
        <div className="bg-white p-6 rounded-lg shadow-sm border border-gray-200">
          <div className="flex items-center justify-between">
            <div>
              <p className="text-sm font-medium text-gray-600">Total vues</p>
              <p className="text-2xl font-bold text-purple-600 mt-1">{totalVues}</p>
            </div>
            <CheckCircle className="h-8 w-8 text-purple-500" />
          </div>
        </div>
      </div>
